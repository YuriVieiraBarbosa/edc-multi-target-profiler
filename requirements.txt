# Python 3.11 or 3.12 (verified on both; 3.12.3 reproduces the predictions from 3.11.16
# with a difference of 0.0 across the 1,098 values of the reference panel).
#
# rdkit and scikit-learn must be EXACTLY these versions: the models (.joblib) are
# serialized with them, and the number of RDKit descriptors defines the feature vector
# (2048 ECFP4 bits + 132 descriptors = 2180). With a different RDKit version the predictor fails
# with "Feature shape mismatch".
rdkit==2026.3.6
scikit-learn==1.9.1

# xgboost: the 21 XGBoost models were trained with 3.4.2, which comes from conda-forge and
# does NOT exist on PyPI (the last one published there is 3.4.1). 3.4.1 was verified: it loads
# the 61 models and reproduces the 1,098 predictions of the panel with a difference of 0.0, emitting only
# a cosmetic warning about configuration generated by a previous version. When installing via conda,
# prefer 3.4.2. The nvidia-nccl-cu13 dependency (305 MB) that pip pulls in along with it only
# serves GPU purposes and can be uninstalled: CPU prediction remains identical.
xgboost==3.4.1

# Versions with which the pipeline was tested
numpy==2.4.6
pandas==3.0.6
scipy==1.17.1
joblib==1.6.0
matplotlib==3.11.2
